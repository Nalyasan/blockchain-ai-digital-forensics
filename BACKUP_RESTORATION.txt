2. VERIFICATION OF BACKUP INTEGRITY
------------------------------------
To independently verify the backup archive against its master SHA-256 manifest:

PowerShell Command:
  $manifest = Get-Content 'D:\Projest_backup_G0\MANIFEST_SHA256.txt'
  $mismatches = 0
  $checked = 0
  foreach ($line in $manifest) {
      if (-not [string]::IsNullOrWhiteSpace($line)) {
          $parts = $line -split '  ', 2
          $expectedHash = $parts[0].Trim().ToUpper()
          $relPath = $parts[1].Trim()
          $fullPath = Join-Path 'D:\Projest_backup_G0' $relPath
          if (Test-Path $fullPath) {
              $actualHash = (Get-FileHash -Path $fullPath -Algorithm SHA256).Hash.ToUpper()
              if ($actualHash -ne $expectedHash) {
                  Write-Error "Hash mismatch for $relPath: expected $expectedHash, got $actualHash"
                  $mismatches++
              }
              $checked++
          } else {
              Write-Error "Missing backup file: $relPath"
              $mismatches++
          }
      }
  }
  Write-Host "Verification complete: $checked files checked, $mismatches mismatches."

Python Command:
  python -c "
  import hashlib
  from pathlib import Path

  backup_root = Path('D:/Projest_backup_G0')
  manifest_path = backup_root / 'MANIFEST_SHA256.txt'
  assert manifest_path.exists(), 'Manifest not found!'

  total, errors = 0, 0
  with open(manifest_path, 'r', encoding='utf-8') as f:
      for line in f:
          line = line.strip()
          if not line:
              continue
          expected_sha, rel_path = line.split('  ', 1)
          target = backup_root / rel_path
          if not target.exists():
              print(f'MISSING: {rel_path}')
              errors += 1
              continue
          h = hashlib.sha256()
          with open(target, 'rb') as tf:
              while chunk := tf.read(65536):
                  h.update(chunk)
          if h.hexdigest().upper() != expected_sha.upper():
              print(f'MISMATCH: {rel_path}')
              errors += 1
          total += 1
  print(f'Checked {total} files: {errors} errors.')
  "
